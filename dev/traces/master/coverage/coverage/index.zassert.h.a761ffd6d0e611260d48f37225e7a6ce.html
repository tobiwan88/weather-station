32 /*
33 * Granular, per-source-file assertions.
34 *
35 * ZASSERT_MODULE() opts a scope into the module-aware ZASSERT() macro. The
36 * resulting assertion level is a compile-time constant, so disabled assertions
37 * (and, below ZASSERT_LEVEL_VERBOSE, the message and its arguments) are folded away by
38 * the optimizer. An individual file may set its level explicitly, overriding the
39 * module default.
40 *
41 * For headers and inline functions, place ZASSERT_MODULE() inside the function
42 * body so the selection does not leak into the includer, or use ZASSERT_TERSE(),
43 * ZASSERT_NORMAL() or ZASSERT_VERBOSE() for an assertion with a fixed level.
44 */
45
46 /** Assertions are disabled and compiled out. */
47 #define ZASSERT_LEVEL_OFF 0
48 /** Assertions are checked; No additional information is reported. */
49 #define ZASSERT_LEVEL_TERSE 1
50 /** Assertions are checked; only the failing location is reported. */
51 #define ZASSERT_LEVEL_NORMAL 2
52 /** Assertions are checked; condition, location and message are reported. */
53 #define ZASSERT_LEVEL_VERBOSE 3

125 /*
126 * ZASSERT_MODULE(module) -> CONFIG_ASSERT_MODULE_<module>_LEVEL
127 * ZASSERT_MODULE(module, level) -> explicit level
128 */
129 #define ZASSERT_IMPL_LEVEL_RESOLVE(...) \
130 GET_ARG_N(2, __VA_ARGS__, ZASSERT_IMPL_MODULE_LEVEL(GET_ARG_N(1, __VA_ARGS__)), _)

132 #ifndef CONFIG_ASSERT
133 #define ZASSERT_IMPL(_level, _test, ...)
134 #define ZASSERT_IMPL_MODULE(...)
135 #else
136 #define ZASSERT_IMPL(_level, _test, ...) \
137 do { \
138 if ((_level) != ZASSERT_LEVEL_OFF && unlikely(!(_test))) { \
139 switch ((_level)) { \
140 case ZASSERT_LEVEL_VERBOSE: \
141 zassert_fail(Z_STRINGIFY(_test), __FILE__, __LINE__, \
142 COND_CODE_1(IS_EMPTY(__VA_ARGS__), \
143 (NULL), (__VA_ARGS__))); \
144 break; \
145 case ZASSERT_LEVEL_NORMAL: \
146 zassert_fail(NULL, __FILE__, __LINE__, NULL); \
147 break; \
148 case ZASSERT_LEVEL_TERSE: \
149 zassert_fail(NULL, NULL, 0, NULL); \
150 break; \
151 default: /* ZASSERT_LEVEL_OFF */ \
152 break; \
153 } \
154 } \
155 } while (false)

199 /**
200 * @brief Select the Kconfig-defined assertion module used by ZASSERT() in this file.
201 *
202 * Must be placed at file scope before any use of ZASSERT() in the translation
203 * unit. Every file that uses ZASSERT() must select a module.
204 *
205 * @param ... The assertion module name (an UPPERCASE identifier) followed by an
206 * optional explicit level. The module's default level is taken from
207 * the Kconfig symbol CONFIG_ASSERT_MODULE_<module>_LEVEL, which must exist
208 * (the name is pasted verbatim, so it must match). An optional
209 * second argument (ZASSERT_LEVEL_OFF, ZASSERT_LEVEL_TERSE,
210 * ZASSERT_LEVEL_NORMAL or ZASSERT_LEVEL_VERBOSE) overrides the module
211 * default for this file.
212 *
213 * @note CONFIG_ASSERT is the master switch: when it is disabled every module is
214 * forced to ZASSERT_LEVEL_OFF, regardless of the module default or an explicit
215 * level.
216 */
217 #define ZASSERT_MODULE(...) ZASSERT_IMPL_MODULE(__VA_ARGS__)
218 /**
219 * @brief Module-aware assertion.
220 *
221 * Gated by the level from ZASSERT_MODULE() in this file. At ZASSERT_LEVEL_TERSE
222 * only a fixed "ASSERTION FAIL" banner is reported. At ZASSERT_LEVEL_NORMAL the
223 * failing location is added. At ZASSERT_LEVEL_VERBOSE the stringified condition
224 * and the optional message and its arguments are also compiled in and printed.
225 *
226 * @param _test Condition to check. A fatal error is raised if it is false.
227 * @param ... Optional message format string followed by its arguments. May be
228 * omitted when the check is self-explanatory.
229 */
230 #define ZASSERT(_test, ...) ZASSERT_IMPL(zassert_impl_level, _test, ##__VA_ARGS__)
