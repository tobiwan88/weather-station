73 /**
74 * @brief Structure for transfer flash sector boundaries
75 *
76 * This template is used for presentation of flash memory structure. It
77 * consumes much less RAM than @ref flash_area.
78 *
79 * This uses ptrdiff_t for fs_off and size_t for fs_size, which limits
80 * flash devices to no larger than the address space of the target as a way
81 * to constrain the size of this struct.
82 */
83 struct flash_sector {
84 /** Sector offset from the beginning of the flash device */
85 ptrdiff_t fs_off;
86 /** Sector size in bytes */
87 size_t fs_size;

238 /**
239 * @brief Erase flash area or fill with erase-value
240 *
241 * On program-erase devices this function behaves exactly like flash_area_erase.
242 * On RAM non-volatile device it will call erase, if driver provides such
243 * callback, or will fill given range with erase-value defined by driver.
244 * This function should be only used by code that has not been written
245 * to directly support devices that do not require erase and rely on
246 * device being erased prior to some operations.
247 * Note that emulated erase, on devices that do not require, is done
248 * via write, which affects endurance of device.
249 *
250 * @see flash_area_erase()
251 * @see flash_flatten()
252 *
253 * @param[in] fa Flash area
254 * @param[in] off Offset relative from beginning of flash area.
255 * @param[in] len Number of bytes to be erase
256 *
257 * @return 0 on success, negative errno code on fail.
258 */
