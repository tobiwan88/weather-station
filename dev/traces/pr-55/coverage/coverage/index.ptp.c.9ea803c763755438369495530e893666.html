173 static void ptp_print_hw_timestamp_formated(const struct shell *sh, struct net_ptp_time ts,
174 const char *name, const char *suffix)
175 {
176 uint32_t ms = ts.nanosecond / NSEC_PER_MSEC;
177 uint32_t us = (ts.nanosecond / NSEC_PER_USEC) % USEC_PER_MSEC;
178 uint32_t ns = ts.nanosecond % NSEC_PER_USEC;
179 struct tm tm_timestamp = {0};
180 time_t time_seconds = ts.second;
181
182 gmtime_r(&time_seconds, &tm_timestamp);
183
184 #if defined(CONFIG_REQUIRES_FULL_LIBC)
185 char time_str[sizeof("1970-01-01 00:00:00")];
186
187 strftime(time_str, sizeof(time_str), "%F %T", &tm_timestamp);
188
189 PR("%s: %s.%03u,%03u,%03u %s\n", name, time_str, ms, us, ns, suffix);
190 #else /* CONFIG_REQUIRES_FULL_LIBC */
191 PR("%s: %04u-%02u-%02u %02u:%02u:%02u.%03u,%03u,%03u %s\n", name,
192 tm_timestamp.tm_year + 1900, tm_timestamp.tm_mon + 1, tm_timestamp.tm_mday,
193 tm_timestamp.tm_hour, tm_timestamp.tm_min, tm_timestamp.tm_sec, ms, us, ns, suffix);
194 #endif /* CONFIG_REQUIRES_FULL_LIBC */
195 }

197 static void ptp_print_hw_timestamp(const struct shell *sh, struct net_if *iface, const char *name)
198 {
199 struct net_ptp_time ts;
200
201 if (ptp_read_hw_timestamp(iface, &ts)) {
202 PR("%s: %" PRIu64 ".%09u\n", name, ts.second, ts.nanosecond);
203
204 ptp_print_hw_timestamp_formated(sh, ts, name, "(TAI)");
205
206 ts.second -= ptp_clock_time_prop_ds()->current_utc_offset; /* TAI-UTC offset */
207
208 ptp_print_hw_timestamp_formated(sh, ts, name, "(UTC)");
209 } else {
210 PR("%s: unavailable\n", name);
211 }

302 {
303 struct ptp_port *port = ptp_find_port(port_id);
304 struct ptp_dataset *best = NULL;
305 char port_clock_id[sizeof("FF:FF:FF:FF:FF:FF:FF:FF")];
306 char best_sender_id[sizeof("FF:FF:FF:FF:FF:FF:FF:FF")];
307
308 if (!port) {
309 PR_WARNING("PTP port %u was not found.\n", port_id);
310 return;
311 }
312
313 ptp_sprint_clock_id(&port->port_ds.id.clk_id, port_clock_id, sizeof(port_clock_id));
314
315 PR("PTP Port %u:\n", port->port_ds.id.port_number);
316 PR("interface : %d\n", net_if_get_by_iface(port->iface));
317 PR("port identity : %s-%u\n", port_clock_id, port->port_ds.id.port_number);
318
319 PR("\nConfiguration:\n");
320 PR("state : %s\n", ptp_port_state2str(ptp_port_state(port)));
321 PR("enabled : %s\n", port->port_ds.enable ? "yes" : "no");
322 PR("protocol : %s\n", ptp_net_protocol2str());
323 PR("time transmitter only : %s\n", port->port_ds.time_transmitter_only ? "yes" : "no");
324 PR("announce log itv : %d\n", port->port_ds.log_announce_interval);
325 PR("announce timeout : %u\n", port->port_ds.announce_receipt_timeout);
326 PR("sync log itv : %d\n", port->port_ds.log_sync_interval);
327 PR("min delay_req log itv : %d\n", port->port_ds.log_min_delay_req_interval);
328 PR("min pdelay_req log itv: %d\n", port->port_ds.log_min_pdelay_req_interval);
329 PR("delay mechanism : %s\n", ptp_delay_mechanism2str(port->port_ds.delay_mechanism));
330 PR("delay asymmetry : %" PRId64 " ns\n",
331 ptp_timeinterval_to_ns(port->port_ds.delay_asymmetry));
332 PR("mean link delay : %" PRId64 " ns\n",
333 ptp_timeinterval_to_ns(port->port_ds.mean_link_delay));
334 PR("neighbor rate ratio : %s (%" PRId64 " ppb)\n",

342 PR("pdelay pending seq : %u\n", port->pdelay_req_sequence_id);
343 PR("timeouts bits : announce=%u delay=%u sync=%u qualification=%u pdelay=%u\n",

345 atomic_test_bit(&port->timeouts, PTP_PORT_TIMER_DELAY_TO),
346 atomic_test_bit(&port->timeouts, PTP_PORT_TIMER_SYNC_TO),
347 atomic_test_bit(&port->timeouts, PTP_PORT_TIMER_QUALIFICATION_TO),
348 atomic_test_bit(&port->timeouts, PTP_PORT_TIMER_PDELAY_TO));
349 PR("timer remaining ms : announce=%d delay=%d sync=%d qualification=%d pdelay=%d\n",
350 k_timer_remaining_get(&port->timers.announce),
351 k_timer_remaining_get(&port->timers.delay), k_timer_remaining_get(&port->timers.sync),
352 k_timer_remaining_get(&port->timers.qualification),
353 k_timer_remaining_get(&port->timers.pdelay));
354 ptp_print_hw_timestamp(sh, port->iface, "PHC now ");
355
356 best = ptp_port_best_foreign_ds(port);
357 if (!best) {
358 PR("best foreign : none\n");
359 return;
360 }
361
362 ptp_sprint_clock_id(&best->sender.clk_id, best_sender_id, sizeof(best_sender_id));
363 PR("best foreign : %s-%u (p1=%u p2=%u steps_rm=%u)\n", best_sender_id,
